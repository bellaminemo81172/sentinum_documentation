Febris CO2/TH/SCW Payload Decoder LoRaWAN®

The following JavaScript decoder can be used in LoRaWAN® network servers such as The Things Stack (TTN) to convert the payloads transmitted by the Febris CO2/TH/SCW sensor into readable measurement values. 

function decodeUplink(input) {
    var decoded = {};
    var bytes = input.bytes;
    
if (input.fPort == 1) {//TELEMETRY
        //decode header
        decoded.base_id = bytes[0] >> 4;
        decoded.major_version = bytes[0] & 0x0F;
        decoded.minor_version = bytes[1] >> 4;
        decoded.product_version = bytes [1] & 0x0F;
        decoded.up_cnt = bytes[2];
        decoded.battery_voltage = ((bytes[3] << 8) | bytes[4]) / 1000.0;
        decoded.internal_temperature = ((bytes[5] << 8) | bytes[6]) / 10 - 100;
        
decoded.networkBaseType = ‘lorawan’;
        decoded.networkSubType = ‘tti’;
       
        var it = 7;
       
        if (decoded.minor_version >= 3) {
            it = 7;
       
            // Humidity is included in all variants
            decoded.humidity = bytes[it++];
               
if (decoded.product_version & 0x01) { // CO2 and pressure are included if subversion bit 0 = 1, otherwise 0
                decoded.pressure = (bytes[it++] << 8 | bytes [it++]);
                decoded.co2_ppm = (bytes[it++] << 8 | bytes[it++]);
            } else {
                it += 4; // Values are 0 for compatibility reasons, so skip them
            }
               
decoded.alarm = bytes[it++];//Alarm level, corresponds to green, yellow, red
   
            //Discard FIFO values (1-byte FIFO size, 1-byte period, 7 bytes per FIFO entry)
            it += 2 + bytes[it] * 7;
               
decoded.dew_point = ((bytes[it++] << 8) | bytes[it++]) / 10 - 100;
           
            // Wall temperature and humidity are included if subversion bit 2 = 1
            if (decoded.product_version & 0x04) {
                
decoded.wall_temperature = ((bytes[it++] << 8) | bytes[it++]) / 10 - 100;
                decoded.therm_temperature = ((bytes[it++] << 8) | bytes[it++]) / 10 - 100;
                decoded.wall_humidity = bytes[it++];
            }
           
}else{
            it = 7;
       
            //Humidity is included in all variants
            decoded.humidity = bytes[it++];
               
if (decoded.product_version & 0x01) { // CO2 and pressure are included if subversion bit 0 = 1, otherwise 0
                decoded.pressure = (bytes[it++] << 8 | bytes[it++]);
                decoded.co2_ppm = (bytes[it++] << 8 | bytes[it++]);
            
} else {
                it += 4; // Values are 0 for compatibility reasons, so skip
            }
               
decoded.alarm = bytes[it++];//Alarm level, corresponds to green, yellow, red
   
            //Discard FIFO values (1-byte FIFO size, 1-byte period, 7 bytes per FIFO entry)
            
it += 2 + bytes[it] * 7;
   
            //Dew point included in all variants since minor version 2 (exception: earlier versions of subversion 2, so check the byte)
            if (decoded.minor_version >= 2 && bytes[it] ) {
             
                decoded.dew_point = bytes[it++] - 100;
            
}
           
            // Wall temperature and humidity included if subversion bit 2 = 1
            if (decoded.product_version & 0x04) {
                decoded.wall_temperature = bytes[it++] - 100;
                decoded.therm_temperature = bytes[it++] - 100;
                
decoded.wall_humidity = bytes[it++];
   
            }
   
   
        }
    }
    return {
        data: decoded,
        warnings: [],
        errors: []
    };
}
}
Note
  • The decoder supports Febris CO₂, TH, and SCW variants.
  • The decoding process accounts for different firmware and product versions.
  • Depending on the product version, additional measurement values such as CO₂, pressure, dew point, or wall moisture are processed.
  • The code is intended for use in LoRaWAN® applications and network servers.
